cmake_minimum_required(VERSION 3.12)
project(CSOPESY_Process_Scheduler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(csopesy_scheduler src/scheduler.cpp)
target_include_directories(csopesy_scheduler PUBLIC src)
target_link_libraries(csopesy_scheduler PUBLIC Threads::Threads)
add_executable(csopesy src/main.cpp)
target_link_libraries(csopesy PRIVATE csopesy_scheduler)
add_executable(csopesy_tests tests/scheduler_tests.cpp)
target_link_libraries(csopesy_tests PRIVATE csopesy_scheduler)
configure_file(config-demo.txt ${CMAKE_CURRENT_BINARY_DIR}/config-demo.txt COPYONLY)
configure_file(config.txt ${CMAKE_CURRENT_BINARY_DIR}/config.txt COPYONLY)
enable_testing()
add_test(NAME scheduler_tests COMMAND csopesy_tests)
